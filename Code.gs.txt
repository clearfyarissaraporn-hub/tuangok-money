/**
 * ถั่วงอก Money → Google Sheets (Apps Script)
 * วางโค้ดนี้ใน ส่วนขยาย → Apps Script ของ Google ชีตที่ต้องการเก็บข้อมูล
 * แล้ว Deploy เป็น Web app (Execute as: Me, Who has access: Anyone)
 */
const KEY = '__KEY__'; // รหัสเชื่อมต่อ ต้องตรงกับในแอป
const TX = 'รายการ';
const HEAD = ['id','วันที่','ปี','เดือน','ประเภท','หมวด','กลุ่มงบ','จำนวนเงิน','จ่ายด้วย','ผูกกับ','ดอกเบี้ย','โน้ต'];
const GROUPS = ['จำเป็น','ผ่อนหนี้','ออม/ลงทุน','พัฒนาตัวเอง','ความสุข'];

function out_(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}
function toDate_(s) {
  const p = String(s).split('-');
  return new Date(+p[0], +p[1] - 1, +p[2]);
}
function fmt_(d) {
  return d instanceof Date ? Utilities.formatDate(d, Session.getScriptTimeZone() || 'Asia/Bangkok', 'yyyy-MM-dd') : String(d || '');
}

function doPost(e) {
  try {
    const d = JSON.parse(e.postData.contents);
    if (d.key !== KEY) return out_({ ok: false, error: 'รหัสเชื่อมต่อไม่ถูกต้อง' });
    const lock = LockService.getScriptLock();
    lock.waitLock(20000);
    try {
      const ss = SpreadsheetApp.getActiveSpreadsheet();
      if (d.action === 'sync') {
        writeTx_(ss, d.rows || []);
        writeTable_(ss, 'หนี้และบัตร', d.debtHead || [], d.debts || []);
        writeTable_(ss, 'ออมและลงทุน', d.saveHead || [], d.saves || []);
        writeSummary_(ss, d.rows || []);
        return out_({ ok: true, rows: (d.rows || []).length, at: new Date().toISOString(), name: ss.getName() });
      }
      return out_({ ok: false, error: 'ไม่รู้จักคำสั่ง' });
    } finally {
      lock.releaseLock();
    }
  } catch (err) {
    return out_({ ok: false, error: String(err) });
  }
}

function doGet(e) {
  const p = (e && e.parameter) || {};
  if (p.key !== KEY) return out_({ ok: false, error: 'รหัสเชื่อมต่อไม่ถูกต้อง' });
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (p.action === 'ping') return out_({ ok: true, name: ss.getName() });
  const rows = readTx_(ss);
  if (p.action === 'tx') return out_({ ok: true, head: HEAD, rows: rows });
  if (p.action === 'summary') return out_({ ok: true, ...summarize_(rows) });
  return out_({ ok: false, error: 'ไม่รู้จักคำสั่ง' });
}

function readTx_(ss) {
  const sh = ss.getSheetByName(TX);
  if (!sh || sh.getLastRow() < 2) return [];
  return sh.getRange(2, 1, sh.getLastRow() - 1, HEAD.length).getValues()
    .filter(r => r[1] !== '' && r[7] !== '')
    .map(r => r.map((c, i) => (i === 1 ? fmt_(c) : c)));
}

function writeTx_(ss, rows) {
  const sh = ss.getSheetByName(TX) || ss.insertSheet(TX, 0);
  sh.clearContents();
  sh.getRange(1, 1, 1, HEAD.length).setValues([HEAD]).setFontWeight('bold').setBackground('#E04FB5').setFontColor('#FFFFFF');
  if (rows.length) {
    const vals = rows.map(r => { const x = r.slice(0, HEAD.length); while (x.length < HEAD.length) x.push(''); x[1] = toDate_(x[1]); return x; });
    sh.getRange(2, 1, vals.length, HEAD.length).setValues(vals);
    sh.getRange(2, 2, vals.length, 1).setNumberFormat('yyyy-mm-dd');
    sh.getRange(2, 8, vals.length, 1).setNumberFormat('#,##0.00');
  }
  sh.setFrozenRows(1);
}

function writeTable_(ss, name, head, rows) {
  const sh = ss.getSheetByName(name) || ss.insertSheet(name);
  sh.clearContents();
  if (!head.length) return;
  sh.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold').setBackground('#E04FB5').setFontColor('#FFFFFF');
  if (rows.length) sh.getRange(2, 1, rows.length, head.length).setValues(rows.map(r => { const x = r.slice(0, head.length); while (x.length < head.length) x.push(''); return x; }));
  sh.setFrozenRows(1);
}

function summarize_(rows) {
  const m = {}, y = {};
  const blank = () => ({ inc: 0, out: 0, debt: 0, save: 0, need: 0, grow: 0, joy: 0 });
  rows.forEach(r => {
    const date = String(r[1]), type = r[4], group = r[6], amt = Number(r[7]) || 0;
    const mk = date.slice(0, 7), yk = date.slice(0, 4);
    [[m, mk], [y, yk]].forEach(([o, k]) => {
      o[k] = o[k] || blank();
      if (type === 'รายรับ' && r[5] !== 'ถอนเงินออม/ขายลงทุน') o[k].inc += amt;
      if (type === 'รายจ่าย') {
        o[k].out += amt;
        if (group === 'ผ่อนหนี้') o[k].debt += amt;
        if (group === 'ออม/ลงทุน') o[k].save += amt;
        if (group === 'จำเป็น') o[k].need += amt;
        if (group === 'พัฒนาตัวเอง') o[k].grow += amt;
        if (group === 'ความสุข') o[k].joy += amt;
      }
    });
  });
  const toList = o => Object.keys(o).sort().map(k => ({ key: k, ...o[k] }));
  return { months: toList(m), years: toList(y), count: rows.length, at: new Date().toISOString() };
}

function writeSummary_(ss, rows) {
  const s = summarize_(rows.map(r => { const x = r.slice(); x[1] = String(x[1]); return x; }));
  const head = ['ช่วง', 'รายรับ', 'รายจ่าย', 'จำเป็น', 'ผ่อนหนี้', 'ออม/ลงทุน', 'พัฒนาตัวเอง', 'ความสุข', 'คงเหลือ', 'อัตราออม'];
  const line = o => [o.key, o.inc, o.out, o.need, o.debt, o.save, o.grow, o.joy, o.inc - o.out, o.inc ? o.save / o.inc : 0];
  [['สรุปรายเดือน', s.months], ['สรุปรายปี', s.years]].forEach(([name, list]) => {
    const sh = ss.getSheetByName(name) || ss.insertSheet(name);
    sh.clearContents();
    sh.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold').setBackground('#E04FB5').setFontColor('#FFFFFF');
    if (list.length) {
      sh.getRange(2, 1, list.length, head.length).setValues(list.map(line));
      sh.getRange(2, 1, list.length, 1).setNumberFormat('@');
      sh.getRange(2, 2, list.length, 8).setNumberFormat('#,##0');
      sh.getRange(2, 10, list.length, 1).setNumberFormat('0%');
    }
    sh.setFrozenRows(1);
  });
}
